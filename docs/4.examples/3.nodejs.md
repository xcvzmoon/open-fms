# Node.js

Use Node.js 26 or later with its built-in `fetch`. Set the [shared environment variables](/examples#before-you-start) and put `report.pdf` in your working directory.

Use an ESM project (`"type": "module"` in `package.json`). Install `valibot` and the development dependencies `typescript` and `@types/node` with your package manager. Node.js 26 runs these `.ts` files directly.

## Create the upload helper

Save this as `fms.ts`. The helper reads the file into memory, so use it for small files. It checks HTTP responses and returns the file ID and scan status from Open FMS.

```ts [fms.ts]
import { readFile } from 'node:fs/promises';
import { basename } from 'node:path';
import * as v from 'valibot';

const createdUploadSchema = v.object({ fileId: v.string() });
const uploadedFileSchema = v.object({
  fileId: v.string(),
  status: v.string(),
  sizeBytes: v.number(),
  checksumSha256: v.string(),
});
export const fileStatusSchema = v.object({ status: v.string() });
export type UploadedFile = v.InferOutput<typeof uploadedFileSchema>;

const { FMS_URL, FMS_API_KEY, BACKEND_ID } = process.env;
if (!FMS_URL || !FMS_API_KEY || !BACKEND_ID) {
  throw new Error('Set FMS_URL, FMS_API_KEY, and BACKEND_ID');
}

export async function fmsRequest(path: string, options: RequestInit = {}): Promise<Response> {
  const headers = new Headers(options.headers);
  headers.set('Authorization', `Bearer ${FMS_API_KEY}`);
  const response = await fetch(`${FMS_URL}${path}`, {
    ...options,
    headers,
    signal: AbortSignal.timeout(120_000),
  });
  if (!response.ok) {
    throw new Error(`Open FMS returned HTTP ${response.status}: ${await response.text()}`);
  }
  return response;
}

export async function uploadFile(filename: string): Promise<UploadedFile> {
  const bytes = await readFile(filename);
  const created = await fmsRequest('/api/v1/uploads', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      storageBackendId: BACKEND_ID,
      originalFilename: basename(filename),
      declaredSizeBytes: bytes.byteLength,
      mode: 'proxy_single',
    }),
  });
  const { fileId } = v.parse(createdUploadSchema, await created.json());
  const uploaded = await fmsRequest(`/api/v1/uploads/${fileId}/content`, {
    method: 'PUT',
    headers: { 'Content-Type': 'application/octet-stream' },
    body: bytes,
  });
  return v.parse(uploadedFileSchema, await uploaded.json());
}
```

## Upload and inspect the file

```ts [upload.ts]
import { fmsRequest, uploadFile } from './fms.ts';

const { fileId } = await uploadFile('report.pdf');
const response = await fmsRequest(`/api/v1/files/${fileId}`);
console.info(await response.json());
```

```bash
node upload.ts
```

The status can still be pending while the scanner works. Save the returned file ID and check it again later.

## Download after scanning

Save this as `download.ts`. It checks the file status before downloading.

```ts [download.ts]
import { writeFile } from 'node:fs/promises';
import * as v from 'valibot';
import { fileStatusSchema, fmsRequest } from './fms.ts';

const fileId = process.argv[2];
if (!fileId) throw new Error('Usage: node download.ts <file-id>');
const path = `/api/v1/files/${encodeURIComponent(fileId)}`;
const metadata = await fmsRequest(path);
const { status } = v.parse(fileStatusSchema, await metadata.json());
if (status !== 'clean') throw new Error(`File is not clean: ${status}`);

const content = await fmsRequest(`${path}/content`);
await writeFile('downloaded-report.pdf', Buffer.from(await content.arrayBuffer()));
```

```bash
node download.ts '<file-id>'
```

The [Express](/examples/express), [Nitro](/examples/nitro), and [Elysia](/examples/elysia) examples reuse this upload helper.
